ORG 0x1F3       ; Начало программы по условию
MODE:       WORD 0x0        ; 0 - старший байт, 1 - младший байт
INDEX:      WORD 0x5A1      ; Адрес начала строки

START:
    JUMP MAIN_LOOP

MAIN_LOOP:
    IN 3            ; Читаем состояние ВУ-1
    AND #0x40       ; Проверка готовности (6-й бит)
    BEQ MAIN_LOOP   ; Если 0, ждем

    LD MODE         ; Проверяем, какую часть слова выводить
    BEQ LOAD_HIGH   ; Если 0 — выводим старший байт
    JUMP LOAD_LOW   ; Если 1 — выводим младший байт

LOAD_HIGH:
    INC             ; Устанавливаем MODE в 1 для следующего шага
    ST MODE
    LD (INDEX)      ; Загружаем слово из памяти
    SWAB            ; Меняем байты местами, чтобы старший стал младшим
    CALL OUTPUT
    JUMP MAIN_LOOP

LOAD_LOW:
    CLA             ; Сбрасываем MODE в 0
    ST MODE
    LD (INDEX)+     ; Загружаем слово и СРАЗУ сдвигаем указатель INDEX
    CALL OUTPUT
    JUMP MAIN_LOOP

OUTPUT:
    OUT 2           ; Вывод символа в регистр данных ВУ-1
    AND #0xFF       ; Оставляем только младший байт для проверки
    BEQ HALT        ; Если это NUL (00), выходим
    RET

HALT:
    HLT

; --- Данные ---
ORG 0x5A1
MESSAGE:    WORD 0xBB00, 0xD2D5, 0xDDEC, 0x3B00
