ORG 0x10

STOP_SYMBOL: WORD 0
SYMBOL_SEQ_NUMB: WORD 0
CURRENT_SYMBOL: WORD 0xFF00

ITER: WORD 23
SSADD: WORD 0x100
SSADDC: WORD 0x100

HALF_WORD: WORD 0


;УСЛОВИЯ
;Кодировка - ISO-8859-5
;Ввод стоп-символа - ВУ3
;Ввод символа - клавиатура
;Вывод символа - ВУ1

DATA_ADDR: WORD 0x5A1

START:
	CLA
	CALL SELECT_STOP_SYMBOL

MAIN_LOOP:
	LD #23
	ST ITER
	LD SSADDC
	ST SSADD
	
	CALL TAKE_SYMBOL
	CALL FIRST_COND
	JUMP MAIN_LOOP
	
SELECT_STOP_SYMBOL:
	IN 7
	AND #0x40
	BEQ SELECT_STOP_SYMBOL
	IN 6
	ST STOP_SYMBOL
	RET

TAKE_SYMBOL:
	IN 0x19
	AND #0x40
	BEQ TAKE_SYMBOL
	IN 0x18
	AND #0xFF
	ST CURRENT_SYMBOL
	CALL STOP_CHECK

STOP_CHECK:
	CMP STOP_SYMBOL
	BEQ HALT
	RET

FIRST_COND:
	LD SYMBOL_SEQ_NUMB
	BEQ ZERO

ONE:
	LD HALF_WORD
	ADD CURRENT_SYMBOL
	ST (DATA_ADDR)+
	LD #0
	ST SYMBOL_SEQ_NUMB
	ST HALF_WORD
	CALL OUTPUT
	RET

ZERO:
	INC
	ST SYMBOL_SEQ_NUMB
	LD HALF_WORD
	ADD CURRENT_SYMBOL
	SWAB
	ST HALF_WORD
	RET

OUTPUT:
	LD (SSADD)+
	CMP CURRENT_SYMBOL
	BEQ OUTT
	LOOP ITER
	JUMP OUTPUT
	RET

OUTT:
	LD CURRENT_SYMBOL
	OUT 2
	RET

	
	
HALT:
	HLT

ORG 0x100
SPEC_SYMBOLS:

WORD 0x002D, 0x0020, 0x003D, 0x005B, 0x005C, 0x005D, 0x003B, 0x0060
